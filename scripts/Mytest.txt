# Example script demonstrating all implemented functions
# This script performs a complete measurement sequence

# Set data file for results
initialize_data_file filename=custom_data.csv
add_note Sample appears normal

# Initial setup - set temperature and wait for stabilization
set_dyna_temp 300 20 fast_settle
wait_for temp_stable 5

# Set initial field to zero
set_dyna_field 0 10 linear
wait_for field_stable 5

# Scan Helmholtz field while measuring Hall voltage
scan_helmholtz_field 0 500 50 50 linear
	wait_for field_stable 15
	measure_hall_field
	close_channel a
	measure_lockin
	open_all_channels
	close_channel b
	measure_lockin
	open_all_channels

# Now scan Dyna field at fixed Helmholtz field
set_helmholtz_field 200 5
scan_dyna_field -1000 1000 200 10 linear
	wait_for field_stable 1
	measure_hall_field
	close_channel a
	measure_lockin
	open_all_channels
	close_channel b
	measure_lockin
	open_all_channels


# Temperature scan at fixed fields
set_dyna_field 500 10 linear
set_helmholtz_field 100 5
scan_dyna_temp 300 400 25 20 fast_settle
	wait_for temp_stable 1
	measure_hall_field
	close_channel a
	measure_lockin
	open_all_channels
	close_channel a
	measure_lockin
	open_all_channels


# Run a sub-script for additional measurements
run_saved_script additional_measurements.txt

# Final cleanup - return to zero field
set_dyna_field 0 10 linear
set_dyna_temp 300 20 fast_settle
set_helmholtz_field 0 5
